// Auction backend retained for settlement/admin compatibility.
// Inventory is authoritative in SQL table custom_black_market_stock.
-	script	CustomBlackMarket	-1,{
OnOpen:
	// FreaokRO commerce hub: this Basic Skill owns both the remote shop-search/market access and Auction entry.
	switch(select("Mercado / Busca de Lojas:Sistema de Leilao:Fechar")) {
	case 1:
		switch(select("Buscar Lojas de Players:Mercado Negro:Voltar")) {
		case 1:
			// Native Search Store in remote mode, across every map. FreokRO extends
			// the vending search to include persistent Market Clones (@autotrade2).
			// Group level 1+ opens remotely; lower groups click the result to travel to it.
			searchstores 255,1,"all";
			end;
		case 3:
			doevent "CustomBlackMarket::OnOpen";
			end;
		}
		break; // case 2 continues into the SQL Black Market stock below.
	case 2:
		doevent "CustomBlackMarket::OnAuction";
		end;
	case 3:
		end;
	}
	.@n = query_sql("SELECT `item_id`,`price`,`stock`,`vip_only` FROM `custom_black_market_stock` WHERE `enabled`=1 AND `stock`>0 ORDER BY `slot` LIMIT 20", .@item, .@price, .@stock, .@vip);
	if (.@n <= 0) { dispbottom "Black Market: no stock available."; end; }
	.@menu$ = "";
	for (.@i = 0; .@i < .@n; ++.@i) {
		if (.@vip[.@i] && !#CUSTOM_VIP) continue;
		.@map[.@visible] = .@i;
		.@menu$ += getitemname(.@item[.@i])+" - "+.@price[.@i]+"z ["+.@stock[.@i]+"]:";
		++.@visible;
	}
	if (!.@visible) { dispbottom "Black Market: no stock available for this account."; end; }
	.@choice = select(.@menu$) - 1;
	.@i = .@map[.@choice];
	if (Zeny < .@price[.@i]) { dispbottom "Black Market: insufficient Zeny."; end; }
	if (!checkweight(.@item[.@i],1)) { dispbottom "Black Market: inventory/weight limit reached."; end; }
	// Conditional update prevents two simultaneous buyers from consuming the final unit.
	query_sql("UPDATE `custom_black_market_stock` SET `stock`=`stock`-1 WHERE `item_id`="+.@item[.@i]+" AND `enabled`=1 AND `stock`>0" ); .@ok=callfunc("F_AuctionSqlRows");
	if (.@ok <= 0) { dispbottom "Black Market: that item just sold out."; end; }
	Zeny -= .@price[.@i];
	getitem .@item[.@i],1;
	query_sql("INSERT INTO `custom_black_market_log` (`char_id`,`item_id`,`price`,`amount`) VALUES ("+getcharid(0)+","+.@item[.@i]+","+.@price[.@i]+",1)");
	dispbottom "Black Market: purchase completed.";
	end;

OnEcommerce:
	// Dedicated remote player-shop search opened by CUSTOM_ECOMMERCE.
	searchstores 255,1,"all";
	end;


OnAuction:
	if (!getbattleflag("funk_master_enable") || !getbattleflag("funk_auction")) {
		dispbottom "[FUNK] Sistema de Leiloes temporariamente desativado.";
		end;
	}
	mes "[Mercado Negro - Leiloes]";
	mes "Leilao persistente do FreokRO.";
	next;
	switch(select("Procurar Leiloes:Meus Lances:Meus Leiloes:Criar Leilao:Historico de Precos:Leiloes Especiais:Voltar")) {
	case 1: doevent "CustomBlackMarket::OnAuctionBrowse"; end;
	case 2: doevent "CustomBlackMarket::OnAuctionMyBids"; end;
	case 3: doevent "CustomBlackMarket::OnAuctionMine"; end;
	case 4: doevent "CustomBlackMarket::OnAuctionCreate"; end;
	case 5: doevent "CustomBlackMarket::OnAuctionHistory"; end;
	case 6: doevent "CustomBlackMarket::OnAuctionSpecial"; end;
	default: doevent "CustomBlackMarket::OnOpen"; end;
	}

OnAuctionBrowse:
	.@n=query_sql("SELECT `auction_id`,`item_id`,`amount`,`refine`,`current_bid`,`start_price`,`buy_now`,TIMESTAMPDIFF(SECOND,NOW(),`ends_at`),`seller_account_id`,`highest_bidder_account_id` FROM `custom_auction` WHERE `status`='ACTIVE' AND `ends_at`>NOW() ORDER BY `ends_at` ASC LIMIT 30",.@aid,.@item,.@amount,.@refine,.@bid,.@start,.@buy,.@left,.@seller,.@leader);
	if (.@n<=0) { mes "[Leiloes]"; mes "Nao existem leiloes ativos."; close; }
	.@menu$="";
	for(.@i=0;.@i<.@n;.@i++) {
		.@price=(.@bid[.@i]>.@start[.@i]? .@bid[.@i]:.@start[.@i]);
		.@menu$+=(.@refine[.@i]?"+"+.@refine[.@i]+" ":"")+getitemname(.@item[.@i])+" x"+.@amount[.@i]+" - "+.@price+"z:";
	}
	.@s=select(.@menu$)-1;
	.@auction=.@aid[.@s];
	// Re-read the selected row. The list may be stale by the time the player clicks it.
	.@ok=query_sql("SELECT `item_id`,`amount`,`refine`,`current_bid`,`start_price`,`buy_now`,TIMESTAMPDIFF(SECOND,NOW(),`ends_at`),`seller_account_id`,`highest_bidder_account_id`,`highest_bidder_char_id`,`highest_bidder_name` FROM `custom_auction` WHERE `auction_id`="+.@auction+" AND `status`='ACTIVE' AND `ends_at`>NOW() LIMIT 1",.@item2,.@amount2,.@refine2,.@bid2,.@start2,.@buy2,.@left2,.@seller2,.@leader2,.@leader_char2,.@leader_name2$);
	if(.@ok<=0){ mes "[Leilao]"; mes "Este leilao nao esta mais ativo."; close; }
	.@current=(.@bid2>.@start2?.@bid2:.@start2);
	mes "[Leilao #"+.@auction+"]";
	mes (.@refine2?"+"+.@refine2+" ":"")+getitemname(.@item2)+" x"+.@amount2;
	mes "Lance atual: "+.@current+" Zeny";
	if(.@buy2>0) mes "Comprar agora: "+.@buy2+" Zeny";
	mes "Tempo restante: "+(.@left2/60)+" minuto(s).";
	if(.@seller2==getcharid(3)){ mes "Este leilao pertence a sua conta."; close; }
	next;
	.@action$="Dar lance"+(.@buy2>0?":Comprar Agora":"")+":Voltar";
	.@action=select(.@action$);
	if(.@action==1){
		// Minimum increment: max(1000z, 5% of the current price), rounded up.
		.@inc=(.@current*5+99)/100;
		if(.@inc<1000) .@inc=1000;
		.@minimum=(.@bid2>0?.@current+.@inc:.@start2);
		mes "[Dar Lance]";
		mes "Lance minimo: ^0055FF"+.@minimum+" Zeny^000000.";
		input .@offer,.@minimum,2000000000;
		if(.@buy2>0 && .@offer>=.@buy2){ mes "Para pagar o valor de Comprar Agora, use a opcao propria."; close; }
		// If the same account already leads, only the delta needs new escrow.
		.@reserve=(.@leader2==getcharid(3) && .@bid2>0 ? .@offer-.@bid2 : .@offer);
		if(.@reserve<=0){ mes "Este lance nao aumenta sua oferta atual."; close; }
		if(Zeny<.@reserve){ mes "Zeny insuficiente para o valor adicional de "+.@reserve+"."; close; }
		mes "Reservar ^FF0000"+.@reserve+" Zeny^000000 neste lance?";
		next; if(select("Sim:Nao")!=1) close;
		// Phase 8 hardening: all SQL-side effects of a bid commit together.
		// Zeny is reserved locally first and restored if ANY SQL step fails.
		Zeny-=.@reserve;
		query_sql("START TRANSACTION");
		query_sql("UPDATE `custom_auction` SET `current_bid`="+.@offer+",`highest_bidder_account_id`="+getcharid(3)+",`highest_bidder_char_id`="+getcharid(0)+",`highest_bidder_name`='"+escape_sql(strcharinfo(0))+"',`bid_count`=`bid_count`+1,`ends_at`=IF(TIMESTAMPDIFF(SECOND,NOW(),`ends_at`)<=120,DATE_ADD(`ends_at`,INTERVAL 2 MINUTE),`ends_at`) WHERE `auction_id`="+.@auction+" AND `status`='ACTIVE' AND `ends_at`>NOW() AND `seller_account_id`<>"+getcharid(3)+" AND `current_bid`="+.@bid2+" AND `highest_bidder_account_id`="+.@leader2); .@changed=callfunc("F_AuctionSqlRows");
		if(.@changed<=0){ query_sql("ROLLBACK"); Zeny+=.@reserve; mes "Outro jogador alterou este leilao antes de voce. Seu Zeny foi devolvido."; close; }
		query_sql("INSERT INTO `custom_auction_bid` (`auction_id`,`account_id`,`char_id`,`char_name`,`amount`,`refunded`) VALUES ("+.@auction+","+getcharid(3)+","+getcharid(0)+",'"+escape_sql(strcharinfo(0))+"',"+.@offer+",0)"); .@sqlok=callfunc("F_AuctionSqlRows");
		if(.@sqlok<=0){ query_sql("ROLLBACK"); Zeny+=.@reserve; mes "Falha ao registrar o lance. Seu Zeny foi devolvido."; close; }
		// Previous leader is released to the persistent auction wallet in the same transaction.
		if(.@bid2>0 && .@leader2>0 && .@leader2!=getcharid(3)){
			query_sql("INSERT INTO `custom_auction_wallet` (`account_id`,`balance`,`refund_char_id`,`refund_char_name`) VALUES ("+.@leader2+","+.@bid2+","+.@leader_char2+",'"+escape_sql(.@leader_name2$)+"') ON DUPLICATE KEY UPDATE `balance`=`balance`+VALUES(`balance`),`refund_char_id`=VALUES(`refund_char_id`),`refund_char_name`=VALUES(`refund_char_name`),`updated_at`=CURRENT_TIMESTAMP"); .@sqlok=callfunc("F_AuctionSqlRows");
			if(.@sqlok<=0){ query_sql("ROLLBACK"); Zeny+=.@reserve; mes "Falha no escrow de reembolso. Seu Zeny foi devolvido."; close; }
			query_sql("UPDATE `custom_auction_bid` SET `refunded`=1 WHERE `auction_id`="+.@auction+" AND `account_id`="+.@leader2+" AND `amount`="+.@bid2+" AND `refunded`=0 ORDER BY `bid_id` DESC LIMIT 1"); .@sqlok=callfunc("F_AuctionSqlRows");
			if(.@sqlok<=0){ query_sql("ROLLBACK"); Zeny+=.@reserve; mes "Falha ao reconciliar o lance anterior. Seu Zeny foi devolvido."; close; }
		}
		query_sql("COMMIT");
		mes "Lance registrado com sucesso.";
		if(.@left2<=120) mes "Anti-sniping: o encerramento foi prorrogado em 2 minutos.";
		close;
	}
	if(.@buy2>0 && .@action==2){
		mes "[Comprar Agora]";
		mes "Preco: ^FF0000"+.@buy2+" Zeny^000000.";
		mes "A compra encerra o leilao imediatamente.";
		next; if(select("Confirmar:Cancelar")!=1) close;
		 .@reserve=(.@leader2==getcharid(3) && .@bid2>0 ? .@buy2-.@bid2 : .@buy2);
		if(.@reserve<=0){ mes "Sua oferta atual ja cobre este valor."; close; }
		if(Zeny<.@reserve){ mes "Zeny insuficiente para completar a compra (faltam "+.@reserve+")."; close; }
		Zeny-=.@reserve;
		// Phase 8: Buy Now state, bid journal and displaced-bid refund are one SQL transaction.
		query_sql("START TRANSACTION");
		query_sql("UPDATE `custom_auction` SET `status`='SETTLEMENT_PENDING',`current_bid`="+.@buy2+",`highest_bidder_account_id`="+getcharid(3)+",`highest_bidder_char_id`="+getcharid(0)+",`highest_bidder_name`='"+escape_sql(strcharinfo(0))+"',`ended_at`=NOW(),`settlement_reason`='BUY_NOW' WHERE `auction_id`="+.@auction+" AND `status`='ACTIVE' AND `ends_at`>NOW() AND `seller_account_id`<>"+getcharid(3)+" AND `current_bid`="+.@bid2+" AND `highest_bidder_account_id`="+.@leader2); .@changed=callfunc("F_AuctionSqlRows");
		if(.@changed<=0){ query_sql("ROLLBACK"); Zeny+=.@reserve; mes "O leilao mudou antes da confirmacao. Seu Zeny foi devolvido."; close; }
		query_sql("INSERT INTO `custom_auction_bid` (`auction_id`,`account_id`,`char_id`,`char_name`,`amount`,`refunded`) VALUES ("+.@auction+","+getcharid(3)+","+getcharid(0)+",'"+escape_sql(strcharinfo(0))+"',"+.@buy2+",0)"); .@sqlok=callfunc("F_AuctionSqlRows");
		if(.@sqlok<=0){ query_sql("ROLLBACK"); Zeny+=.@reserve; mes "Falha ao registrar Comprar Agora. Seu Zeny foi devolvido."; close; }
		if(.@bid2>0 && .@leader2>0 && .@leader2!=getcharid(3)){
			query_sql("INSERT INTO `custom_auction_wallet` (`account_id`,`balance`,`refund_char_id`,`refund_char_name`) VALUES ("+.@leader2+","+.@bid2+","+.@leader_char2+",'"+escape_sql(.@leader_name2$)+"') ON DUPLICATE KEY UPDATE `balance`=`balance`+VALUES(`balance`),`refund_char_id`=VALUES(`refund_char_id`),`refund_char_name`=VALUES(`refund_char_name`),`updated_at`=CURRENT_TIMESTAMP"); .@sqlok=callfunc("F_AuctionSqlRows");
			if(.@sqlok<=0){ query_sql("ROLLBACK"); Zeny+=.@reserve; mes "Falha no escrow de reembolso. Seu Zeny foi devolvido."; close; }
			query_sql("UPDATE `custom_auction_bid` SET `refunded`=1 WHERE `auction_id`="+.@auction+" AND `account_id`="+.@leader2+" AND `amount`="+.@bid2+" AND `refunded`=0 ORDER BY `bid_id` DESC LIMIT 1"); .@sqlok=callfunc("F_AuctionSqlRows");
			if(.@sqlok<=0){ query_sql("ROLLBACK"); Zeny+=.@reserve; mes "Falha ao reconciliar o lance anterior. Seu Zeny foi devolvido."; close; }
		}
		query_sql("COMMIT");
		mes "Compra reservada com sucesso.";
		mes "O leilao entrou em liquidacao persistente.";
		mes "A entrega do item e o pagamento do vendedor sao processados pela etapa de settlement.";
		close;
	}
	doevent "CustomBlackMarket::OnAuction"; end;

OnAuctionMyBids:
	.@wallet=0; query_sql("SELECT `balance` FROM `custom_auction_wallet` WHERE `account_id`="+getcharid(3),.@wallet);
	.@n=query_sql("SELECT `auction_id`,`item_id`,`current_bid`,`ends_at` FROM `custom_auction` WHERE `status`='ACTIVE' AND `highest_bidder_account_id`="+getcharid(3)+" AND `ends_at`>NOW() ORDER BY `ends_at` LIMIT 30",.@aid,.@item,.@bid,.@end$);
	mes "[Meus Lances]";
	if(.@wallet>0) mes "Reembolso em processamento RodEx: ^0055FF"+.@wallet+" Zeny^000000.";
	if(.@n<=0){ mes "Voce nao lidera nenhum leilao ativo."; close; }
	for(.@i=0;.@i<.@n;.@i++) mes "#"+.@aid[.@i]+" - "+getitemname(.@item[.@i])+" - "+.@bid[.@i]+"z";
	close;

OnAuctionMine:
	.@n=query_sql("SELECT `auction_id`,`item_id`,`current_bid`,`start_price`,`ends_at` FROM `custom_auction` WHERE `status`='ACTIVE' AND `seller_account_id`="+getcharid(3)+" AND `ends_at`>NOW() ORDER BY `ends_at` LIMIT 30",.@aid,.@item,.@bid,.@start,.@end$);
	mes "[Meus Leiloes]";
	if(.@n<=0){ mes "Voce nao possui leiloes ativos."; close; }
	for(.@i=0;.@i<.@n;.@i++) mes "#"+.@aid[.@i]+" - "+getitemname(.@item[.@i])+" - "+(.@bid[.@i]>.@start[.@i]?.@bid[.@i]:.@start[.@i])+"z";
	close;

OnAuctionCreate:
	// Phase 5 economy: VIP changes listing capacity and listing fee, never bid priority.
	// Free accounts pay 27% of the starting value; VIP and higher pay 7%.
	.@listing_fee_rate=(getgroupid()>=1 ? 7 : 27);
	.@max_active=(getgroupid()>=1 ? 10 : 3);
	.@active_count=0;
	query_sql("SELECT COUNT(*) FROM `custom_auction` WHERE `seller_account_id`="+getcharid(3)+" AND `status` IN ('ESCROW_PENDING','ACTIVE','SETTLEMENT_PENDING')",.@active_count);
	if(.@active_count>=.@max_active){ mes "[Criar Leilao]"; mes "Limite de "+.@max_active+" leiloes simultaneos atingido."; close; }
	mes "[Criar Leilao]";
	mes "Escolha um item negociavel do inventario para colocar em escrow.";
	mes "O item sera removido do personagem e ficara reservado no banco ate o encerramento.";
	next;
	getinventorylist;
	.@menu$="";
	.@visible=0;
	for(.@i=0;.@i<@inventorylist_count;.@i++) {
		// Do not auction equipped, bound, rental or non-tradable items.
		if(@inventorylist_equip[.@i] || @inventorylist_bound[.@i] || @inventorylist_expire[.@i] || !@inventorylist_tradable[.@i]) continue;
		.@slot[.@visible]=.@i;
		.@menu$+=( @inventorylist_refine[.@i] ? "+"+@inventorylist_refine[.@i]+" " : "" )+getitemname(@inventorylist_id[.@i])+" x"+@inventorylist_amount[.@i]+":";
		++.@visible;
		if(.@visible>=30) break;
	}
	if(!.@visible){ mes "Voce nao possui itens elegiveis para leilao."; close; }
	.@pick=select(.@menu$)-1;
	.@i=.@slot[.@pick];
	.@amount=@inventorylist_amount[.@i];
	if(.@amount>1){ mes "Quantidade (1 a "+.@amount+"):"; input .@amount,1,@inventorylist_amount[.@i]; }
	mes "Lance inicial em Zeny:"; input .@start,1,2000000000;
	// Round up so every positive listing has at least one Zeny of fee.
	.@listing_fee=(.@start*.@listing_fee_rate+99)/100;
	if(Zeny<.@listing_fee){
		mes "Taxa de anuncio: "+.@listing_fee_rate+"% do valor inicial ("+.@listing_fee+" Zeny).";
		mes "Zeny insuficiente.";
		close;
	}
	mes "Comprar Agora (0 para desativar):"; input .@buy,0,2000000000;
	if(.@buy>0 && .@buy<=.@start){ mes "Comprar Agora precisa ser maior que o lance inicial."; close; }
	mes "Duracao:";
	switch(select("6 horas:12 horas:24 horas:48 horas")){ case 1:.@hours=6;break; case 2:.@hours=12;break; case 3:.@hours=24;break; default:.@hours=48; }
	mes "Item: "+getitemname(@inventorylist_id[.@i])+" x"+.@amount;
	mes "Lance inicial: "+.@start+" Zeny";
	if(.@buy) mes "Comprar Agora: "+.@buy+" Zeny";
	mes "Duracao: "+.@hours+"h";
	mes "Taxa de anuncio ("+.@listing_fee_rate+"%): ^FF0000"+.@listing_fee+" Zeny^000000.";
	mes "Venda concluida: ^FF00003% de imposto^000000 sobre o valor final.";
	mes "Confirmar?";
	next;
	if(select("Sim:Nao")!=1) close;

	// Journal first: ESCROW_PENDING is never visible/biddable. If the process dies
	// during the hand-off, the exact item snapshot remains quarantined in SQL for recovery.
	.@sql$="INSERT INTO `custom_auction` (`seller_account_id`,`seller_char_id`,`item_id`,`amount`,`identify`,`refine`,`attribute`,`bound`,`expire_time`,`unique_id`,`enchantgrade`,`durability`,`card0`,`card1`,`card2`,`card3`,`opt_id0`,`opt_val0`,`opt_par0`,`opt_id1`,`opt_val1`,`opt_par1`,`opt_id2`,`opt_val2`,`opt_par2`,`opt_id3`,`opt_val3`,`opt_par3`,`opt_id4`,`opt_val4`,`opt_par4`,`start_price`,`buy_now`,`ends_at`,`status`) VALUES ("+
		getcharid(3)+","+getcharid(0)+","+@inventorylist_id[.@i]+","+.@amount+","+@inventorylist_identify[.@i]+","+@inventorylist_refine[.@i]+","+@inventorylist_attribute[.@i]+","+@inventorylist_bound[.@i]+","+@inventorylist_expire[.@i]+","+@inventorylist_unique_id[.@i]+","+@inventorylist_enchantgrade[.@i]+","+@inventorylist_durability[.@i]+","+@inventorylist_card1[.@i]+","+@inventorylist_card2[.@i]+","+@inventorylist_card3[.@i]+","+@inventorylist_card4[.@i]+","+@inventorylist_option_id1[.@i]+","+@inventorylist_option_value1[.@i]+","+@inventorylist_option_parameter1[.@i]+","+@inventorylist_option_id2[.@i]+","+@inventorylist_option_value2[.@i]+","+@inventorylist_option_parameter2[.@i]+","+@inventorylist_option_id3[.@i]+","+@inventorylist_option_value3[.@i]+","+@inventorylist_option_parameter3[.@i]+","+@inventorylist_option_id4[.@i]+","+@inventorylist_option_value4[.@i]+","+@inventorylist_option_parameter4[.@i]+","+@inventorylist_option_id5[.@i]+","+@inventorylist_option_value5[.@i]+","+@inventorylist_option_parameter5[.@i]+","+.@start+","+.@buy+",DATE_ADD(NOW(),INTERVAL "+.@hours+" HOUR),'ESCROW_PENDING')";
	query_sql(.@sql$); .@ok=callfunc("F_AuctionSqlRows");
	if(.@ok<=0){ mes "Nao foi possivel iniciar o escrow. Nenhum item foi removido."; close; }
	query_sql("SELECT LAST_INSERT_ID()",.@auction_id);
	if(.@auction_id<=0){ mes "Falha ao obter o identificador do escrow. Nenhum item foi removido."; close; }

	// Remove by inventory index, so cards/options/refine cannot make us delete a different instance.
	if(!delitemidx(@inventorylist_idx[.@i],.@amount)) {
		query_sql("DELETE FROM `custom_auction` WHERE `auction_id`="+.@auction_id+" AND `status`='ESCROW_PENDING'");
		mes "O item mudou no inventario antes da confirmacao. Operacao cancelada.";
		close;
	}
	query_sql("UPDATE `custom_auction` SET `status`='ACTIVE' WHERE `auction_id`="+.@auction_id+" AND `status`='ESCROW_PENDING'"); .@ok=callfunc("F_AuctionSqlRows");
	if(.@ok<=0) {
		// Fail closed: the escrow row remains authoritative/quarantined instead of becoming biddable.
		mes "O leilao ficou em recuperacao segura (#"+.@auction_id+").";
		mes "O item nao sera vendido ate o escrow ser reconciliado.";
		close;
	}
	if(Zeny<.@listing_fee){
		query_sql("UPDATE `custom_auction` SET `status`='ACTIVE',`ends_at`=NOW(),`ended_at`=NOW(),`settlement_reason`='FEE_FAILED' WHERE `auction_id`="+.@auction_id+" AND `status`='ACTIVE'");
		mes "Seu Zeny mudou antes da publicacao. O item entrou em devolucao segura pelo RodEx."; close;
	}
	Zeny-=.@listing_fee;
	query_sql("UPDATE `custom_auction` SET `listing_fee`="+.@listing_fee+" WHERE `auction_id`="+.@auction_id);
	mes "Leilao #"+.@auction_id+" criado com sucesso.";
	mes "O item esta agora em escrow persistente.";
	close;

OnAuctionHistory:
	mes "[Historico de Precos]"; mes "Informe o ID do item:"; input .@item; if(.@item<=0) close;
	query_sql("SELECT COUNT(*),IFNULL(ROUND(AVG(`final_price`)),0),IFNULL(MIN(`final_price`),0),IFNULL(MAX(`final_price`),0) FROM `custom_auction_history` WHERE `item_id`="+.@item+" AND `sold_at`>=DATE_SUB(NOW(),INTERVAL 30 DAY)",.@count,.@avg,.@min,.@max);
	mes "["+getitemname(.@item)+"]";
	if(.@count>0){ mes "Ultimos 30 dias: "+.@count+" venda(s)."; mes "Media: "+.@avg+"z | Min: "+.@min+"z | Max: "+.@max+"z"; } else mes "Sem vendas nos ultimos 30 dias.";
	next;
	.@n=query_sql("SELECT `final_price`,`amount`,`refine`,`sold_at` FROM `custom_auction_history` WHERE `item_id`="+.@item+" ORDER BY `sold_at` DESC LIMIT 10",.@price,.@amount,.@refine,.@date$);
	if(.@n<=0){ mes "Ainda nao existem vendas registradas para este item."; close; }
	mes "Ultimas vendas:"; for(.@i=0;.@i<.@n;.@i++) mes (.@refine[.@i]?"+"+.@refine[.@i]+" ":"")+"x"+.@amount[.@i]+" - "+.@price[.@i]+"z - "+.@date$[.@i];
	close;

OnAuctionSpecial:
	.@n=query_sql("SELECT `auction_id`,`item_id`,`amount`,`refine`,`current_bid`,`start_price`,`buy_now`,TIMESTAMPDIFF(SECOND,NOW(),`ends_at`),`server_note` FROM `custom_auction` WHERE `status`='ACTIVE' AND `is_server`=1 AND `ends_at`>NOW() ORDER BY `ends_at` LIMIT 30",.@aid,.@item,.@amount,.@refine,.@bid,.@start,.@buy,.@left,.@note$);
	mes "[Leiloes Especiais]";
	mes "Itens oficiais do servidor. Todo Zeny vencedor sai da economia.";
	if(.@n<=0){ mes "Nao ha leilao especial ativo neste momento."; close; }
	.@menu$="";
	for(.@i=0;.@i<.@n;.@i++){
		.@price=(.@bid[.@i]>.@start[.@i]?.@bid[.@i]:.@start[.@i]);
		.@menu$+="#"+.@aid[.@i]+" "+(.@refine[.@i]?"+"+.@refine[.@i]+" ":"")+getitemname(.@item[.@i])+" x"+.@amount[.@i]+" - "+.@price+"z:";
	}
	.@s=select(.@menu$)-1;
	mes "[Leilao Especial #"+.@aid[.@s]+"]";
	mes (.@refine[.@s]?"+"+.@refine[.@s]+" ":"")+getitemname(.@item[.@s])+" x"+.@amount[.@s];
	mes "Lance atual: "+(.@bid[.@s]>.@start[.@s]?.@bid[.@s]:.@start[.@s])+" Zeny";
	if(.@buy[.@s]>0) mes "Comprar Agora: "+.@buy[.@s]+" Zeny";
	mes "Tempo restante: "+(.@left[.@s]/60)+" minuto(s).";
	if(.@note$[.@s]!="") mes "Evento: "+.@note$[.@s];
	mes "Para participar, abra ^0055FFProcurar Leiloes^000000 e selecione o leilao #"+.@aid[.@s]+".";
	close;

OnTravelConfirm:
	// Free-account travel confirmation. VIP/group level 1+ never reaches this path.
	if (@BM_TRAVEL_MAP$ == "" || @BM_TRAVEL_X <= 0 || @BM_TRAVEL_Y <= 0) {
		dispbottom "Mercado Negro: destino da loja invalido.";
		end;
	}
	mes "[Mercado Negro]";
	mes "Loja localizada em ^0055FF"+@BM_TRAVEL_MAP$+" ("+@BM_TRAVEL_X+","+@BM_TRAVEL_Y+")^000000.";
	if (@BM_TRAVEL_FEE_TYPE == 1)
		mes "Custo da viagem: ^FF0000"+@BM_TRAVEL_ZENY+" Zeny^000000.";
	else if (@BM_TRAVEL_FEE_TYPE == 2)
		mes "Custo da viagem: ^FF0000"+@BM_TRAVEL_AMOUNT+"x "+getitemname(@BM_TRAVEL_ITEM)+"^000000.";
	else
		mes "Custo da viagem: ^00AA00Gratis^000000.";
	mes "Deseja se mover ate a loja?";
	next;
	if (select("Sim:Nao") != 1) {
		close;
	}
	// Free accounts: 10 successful travels per daily reset. The authoritative
	// counter lives in SQL, not RAM/account-reg cache, so map crashes/restarts
	// cannot renew the quota. CURDATE() makes the reset occur at 00:00 DB/server time.
	.@quota_rows = query_sql("SELECT IF(`reset_date`=CURDATE(),`free_count`,0) FROM `custom_black_market_quota` WHERE `account_id`="+getcharid(3)+" LIMIT 1", .@free_count);
	if (.@quota_rows < 0) {
		mes "[Mercado Negro]";
		mes "Nao foi possivel validar sua cota. Tente novamente.";
		close;
	}
	if (.@free_count >= 10) {
		mes "[Mercado Negro]";
		mes "Voce atingiu o limite gratuito de ^FF000010 viagens^000000 neste reset diario.";
		mes "Sua cota sera renovada no proximo reset.";
		close;
	}
	if (@BM_TRAVEL_FEE_TYPE == 1) {
		if (Zeny < @BM_TRAVEL_ZENY) {
			mes "[Mercado Negro]";
			mes "Voce nao possui Zeny suficiente para a viagem.";
			close;
		}
		Zeny -= @BM_TRAVEL_ZENY;
	} else if (@BM_TRAVEL_FEE_TYPE == 2) {
		if (countitem(@BM_TRAVEL_ITEM) < @BM_TRAVEL_AMOUNT) {
			mes "[Mercado Negro]";
			mes "Voce nao possui os itens necessarios para a viagem.";
			close;
		}
		delitem @BM_TRAVEL_ITEM,@BM_TRAVEL_AMOUNT;
	}
	// Persist the use immediately before the warp. A restart/crash after this point
	// does not refund the quota, preventing restart abuse.
	query_sql("INSERT INTO `custom_black_market_quota` (`account_id`,`reset_date`,`free_count`) VALUES ("+getcharid(3)+",CURDATE(),1) ON DUPLICATE KEY UPDATE `free_count`=IF(`reset_date`=CURDATE(),`free_count`+1,1), `vip_count`=IF(`reset_date`=CURDATE(),`vip_count`,0), `reset_date`=CURDATE(), `updated_at`=CURRENT_TIMESTAMP"); .@quota_save=callfunc("F_AuctionSqlRows");
	if (.@quota_save < 0) {
		// Fail closed. The fee was already validated; refund it if persistence fails.
		if (@BM_TRAVEL_FEE_TYPE == 1) Zeny += @BM_TRAVEL_ZENY;
		else if (@BM_TRAVEL_FEE_TYPE == 2) getitem @BM_TRAVEL_ITEM,@BM_TRAVEL_AMOUNT;
		mes "[Mercado Negro]";
		mes "Nao foi possivel salvar sua cota. A viagem foi cancelada e o custo devolvido.";
		close;
	}
	.@map$ = @BM_TRAVEL_MAP$;
	.@x = @BM_TRAVEL_X;
	.@y = @BM_TRAVEL_Y;
	@BM_TRAVEL_MAP$ = "";
	@BM_TRAVEL_X = @BM_TRAVEL_Y = 0;
	@BM_TRAVEL_FEE_TYPE = @BM_TRAVEL_ZENY = @BM_TRAVEL_ITEM = @BM_TRAVEL_AMOUNT = 0;
	close2;
	warp .@map$,.@x,.@y;
	end;

}
